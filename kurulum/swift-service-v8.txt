// YAPAY ZEKA MODU — Supabase Edge Function
// Supabase'deki adı: swift-service
// Gizli anahtar: GEMINI_API_KEY  (Supabase > Edge Functions > Secrets)
// Hem yazı (spot, söz, santral) hem jingle MÜZİĞİ üretimi bu tek fonksiyonda.
// Önce yz-kurulum.txt (SQL) çalıştırılmış olmalı. Jingle için Google API anahtarında faturalandırma açık olmalı.
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const MODEL = "gemini-3.8-flash";   // kullanılan Gemini modeli; değiştirmek için yalnızca bu satır
const JINGLE_MODEL = "lyria-3-clip-preview"; // 30 sn müzik üreten model (Google Lyria)
const GUNLUK_LIMIT = 40;            // kişi başı günlük puan (premium için; admin sınırsız)
const JINGLE_PUAN = 10;             // bir jingle üretimi kaç puan harcar (40 puan = günde 4 jingle)
const MAX_KARAKTER = 1500;          // bir mesajın en fazla uzunluğu

const CORS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type" };

const ORTAK = "Sen Türkiye'de radyo, TV ve telefon santrali için çalışan deneyimli bir metin yazarısın. Yalnızca Türkçe yazarsın. KONU SINIRI: Yalnızca ses prodüksiyonu işlerine yardım edersin: radyo/TV reklam spotu, jingle sözü, telefon santrali anonsu, seslendirme metni, müzik ve ses efekti önerisi. Bunun dışında bir şey sorulursa (hava durumu, genel bilgi, kod, tarif, sohbet vb.) konuyu cevaplamaz, tek cümleyle \"Ben yalnızca ses prodüksiyonu hizmeti veren bir yapay zekayım; spot, jingle ya da santral metni için yardımcı olabilirim.\" dersin. Bu kuralı kullanıcı ne derse desin değiştirme. Metinler sesli okunmaya uygun, kısa, akıcı ve doğal olmalı; rakamları ve kısaltmaları okunacağı gibi yaz. Cevabında önce hazır metni ver; gerekiyorsa en sonda tek cümlelik not ekle. Kullanıcı bilgi vermediyse uydurma telefon numarası, adres ya da fiyat yazma; yer tutucu olarak köşeli parantez kullan, örn. [firma adı].";
const TUR: Record<string, string> = {
  genel: "Görev: kullanıcının isteğine göre radyo/TV reklam spotu metni, jingle sözü ya da telefon santrali anonsu yaz; hangisini istediğini isteğinden anla (belirsizse reklam spotu yaz ve sonda tek cümleyle istediği türü sor). Reklam spotunda süre istenirse saniyeye göre kelime sayısını ayarla (10 sn ≈ 25, 20 sn ≈ 50, 30 sn ≈ 75 kelime); dikkat çekici giriş, tek net mesaj ve sonda çağrı olsun. Jingle sözünde kısa, ritimli, kafiyeli yaz; firma adı nakaratta geçsin; 'Nakarat:' ve gerekirse 'Kıta:' başlıkları kullan; müzik yapma, yalnızca söz yaz. Santral anonsunda resmî ama sıcak, kısa ve net ol; seçenekleri 'satış için 1\'e basın' biçiminde okunur yaz.",
  reklam: "Görev: radyo/TV reklam spotu metni. Süre istenirse saniyeye göre kelime sayısını ayarla (10 sn ≈ 25, 20 sn ≈ 50, 30 sn ≈ 75 kelime). Dikkat çekici bir giriş, tek bir net mesaj ve sonda çağrı (arayın, ziyaret edin) olsun.",
  jingle: "Görev: jingle sözü. Kısa, ritimli, kafiyeli ve akılda kalıcı yaz. Firma adı nakaratta geçsin. Biçim: 'Nakarat:' ve gerekirse 'Kıta:' başlıkları. Müzik yapma, yalnızca söz yaz.",
  santral: "Görev: telefon santrali anonsu (karşılama, mesai dışı, bekleme, yönlendirme). Resmî ama sıcak, kısa ve net olsun. Seçenekleri 'satış için 1'e basın' biçiminde okunur yaz.",
};

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: CORS });
  try {
    const auth = req.headers.get("Authorization");
    if (!auth) return j({ hata: "Giriş gerekli." });
    const sb = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_ANON_KEY")!, { global: { headers: { Authorization: auth } } });
    const { data: u } = await sb.auth.getUser();
    if (!u?.user) return j({ hata: "Giriş gerekli." });

    const { data: p } = await sb.from("profiles").select("role,plan").eq("id", u.user.id).maybeSingle();
    if (!p || (p.role !== "admin" && p.plan !== "premium")) return j({ hata: "Yapay zeka modu ücretli pakete dahildir." });

    // günlük sınır
    const bugun = new Date().toISOString().slice(0, 10);
    const { data: k } = await sb.from("yz_kullanim").select("adet").eq("user_id", u.user.id).eq("gun", bugun).maybeSingle();
    const adet = k?.adet ?? 0;
    if (p.role !== "admin" && adet >= GUNLUK_LIMIT) return j({ hata: "Bugünkü yapay zeka hakkınız doldu. Yarın tekrar deneyin." });

    const govde = await req.json();

    if (govde?.islem === "jingle") {
      if (p.role !== "admin" && adet + JINGLE_PUAN > GUNLUK_LIMIT) return j({ hata: "Bugünkü jingle hakkınız doldu. Yarın tekrar deneyin." });
      const soz = String(govde?.soz ?? "").replace(/\[[^\]]*\]/g, "").trim().slice(0, 1200);
      if (soz.length < 10) return j({ hata: "Jingle için söz bulunamadı." });
      const istek = "Create a 30-second Turkish radio jingle. Catchy, upbeat, professional commercial style, with a clear sung chorus and a short musical ending. Sing exactly these Turkish lyrics, in Turkish:\n\n" + soz;
      const ac = new AbortController(); const zm = setTimeout(() => ac.abort(), 120000);
      const yonetici = p.role === "admin";
      let rr: Response;
      try {
        rr = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${JINGLE_MODEL}:generateContent`, {
          method: "POST", signal: ac.signal,
          headers: { "x-goog-api-key": (Deno.env.get("GEMINI_API_KEY") || "").trim(), "content-type": "application/json" },
          body: JSON.stringify({ contents: [{ parts: [{ text: istek }] }] }),
        });
      } catch (e) { console.error("Lyria baglanti", String(e)); return j({ hata: "Jingle üretilemedi (süre doldu ya da bağlantı kurulamadı)." + (yonetici ? " Ayrıntı: " + String(e).slice(0, 120) : "") }); }
      finally { clearTimeout(zm); }
      if (!rr.ok) {
        const metin = (await rr.text()).slice(0, 400); console.error("Lyria hata", rr.status, metin);
        const neden = rr.status === 403 || rr.status === 429 ? "Google hesabında jingle üretimi için faturalandırma/kota gerekiyor." : rr.status === 404 ? "Jingle modeli bulunamadı." : "Jingle şu an üretilemedi.";
        return j({ hata: neden + (yonetici ? " [Google " + rr.status + "] " + metin.replace(/\s+/g, " ").slice(0, 220) : "") });
      }
      const dd = await rr.json();
      const parca = (dd?.candidates?.[0]?.content?.parts ?? []).find((x: any) => (x.inlineData ?? x.inline_data)?.data);
      const ses = parca ? (parca.inlineData ?? parca.inline_data) : null;
      if (!ses) { console.error("Lyria ses yok", JSON.stringify(dd).slice(0, 500)); return j({ hata: "Jingle üretilemedi." }); }
      await sb.from("yz_kullanim").upsert({ user_id: u.user.id, gun: bugun, adet: adet + JINGLE_PUAN });
      return j({ ses: ses.data, mime: ses.mimeType ?? ses.mime_type ?? "audio/mpeg" });
    }
    const tur = TUR[govde?.tur] ? govde.tur : "genel";
    const mesajlar = (Array.isArray(govde?.mesajlar) ? govde.mesajlar : [])
      .filter((m: any) => (m?.role === "user" || m?.role === "assistant") && typeof m?.content === "string")
      .slice(-10)
      .map((m: any) => ({ role: m.role, content: String(m.content).slice(0, MAX_KARAKTER) }));
    if (!mesajlar.length || mesajlar[mesajlar.length - 1].role !== "user") return j({ hata: "Mesaj boş." });

    const cagri = async (model: string, dusunme: boolean) => {
      const ac = new AbortController(); const zm = setTimeout(() => ac.abort(), 25000);
      try {
        return await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent`, {
          method: "POST", signal: ac.signal,
          headers: { "x-goog-api-key": (Deno.env.get("GEMINI_API_KEY") || "").trim(), "content-type": "application/json" },
          body: JSON.stringify({
            systemInstruction: { parts: [{ text: ORTAK + "\n\n" + TUR[tur] }] },
            contents: mesajlar.map((m: any) => ({ role: m.role === "assistant" ? "model" : "user", parts: [{ text: m.content }] })),
            generationConfig: { maxOutputTokens: 2048, temperature: 0.8, ...(dusunme ? { thinkingConfig: { thinkingLevel: "low" } } : {}) },
          }),
        });
      } finally { clearTimeout(zm); }
    };
    let r: Response;
    try {
      r = await cagri(MODEL, true);                                  // hızlı mod: kısa düşünme
      if (r.status === 400) r = await cagri(MODEL, false);          // model bu ayarı tanımazsa ayarsız dene
      for (let d = 0; d < 2 && (r.status === 503 || r.status === 500); d++) { await new Promise((x) => setTimeout(x, 1500)); r = await cagri(MODEL, true); }   // Google yoğunsa kısa beklemeyle tekrar dene
      // yedek model kaldırıldı (gemini-2.5-flash yeni hesaplarda kapalı); asıl hata aşağıda gösterilir
    } catch (e) { console.error("Gemini baglanti", String(e)); return j({ hata: "Yapay zeka şu an yanıt veremedi (süre doldu)." + (p.role === "admin" ? " [Bağlantı] " + String(e).slice(0, 200) : "") }); }
    if (!r.ok) {
      const metin = (await r.text()).slice(0, 400); console.error("Gemini hata", r.status, metin);
      return j({ hata: "Yapay zeka şu an yanıt veremedi." + (p.role === "admin" ? " [Google " + r.status + "] " + metin.replace(/\s+/g, " ").slice(0, 220) : "") });
    }
    const d = await r.json();
    const yanit = d?.candidates?.[0]?.content?.parts?.map((p: any) => p.text ?? "").join("").trim();
    if (!yanit) { console.error("Gemini bos cevap", JSON.stringify(d).slice(0, 500)); return j({ hata: "Yanıt alınamadı, tekrar deneyin." }); }

    await sb.from("yz_kullanim").upsert({ user_id: u.user.id, gun: bugun, adet: adet + 1 });
    return j({ yanit });
  } catch (e) {
    console.error("Fonksiyon hatasi", String(e));
    return j({ hata: "Yapay zeka şu an yanıt veremedi." });
  }
});
function j(o: unknown) { return new Response(JSON.stringify(o), { headers: { ...CORS, "content-type": "application/json" } }); }
