// YAPAY ZEKA MODU — Supabase Edge Function (ileride kurulacak, şimdilik hazır bekliyor)
// Adı: yz-sohbet   |  Gizli anahtar: ANTHROPIC_API_KEY (Supabase > Edge Functions > Secrets)
// Kurulumu birlikte yapacağız; bu dosyayı şimdilik hiçbir yere yüklemeyin.
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";
const CORS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type" };
Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: CORS });
  try {
    const sb = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_ANON_KEY")!, { global: { headers: { Authorization: req.headers.get("Authorization")! } } });
    const { data: u } = await sb.auth.getUser(); if (!u?.user) return j({ hata: "Giriş gerekli." });
    const { data: p } = await sb.from("profiles").select("role,plan").eq("id", u.user.id).maybeSingle();
    if (!p || (p.role !== "admin" && p.plan !== "premium")) return j({ hata: "Yapay zeka modu ücretli pakete dahildir." });
    const { mesajlar } = await req.json();
    const r = await fetch("https://api.anthropic.com/v1/messages", {
      method: "POST",
      headers: { "x-api-key": Deno.env.get("ANTHROPIC_API_KEY")!, "anthropic-version": "2023-06-01", "content-type": "application/json" },
      body: JSON.stringify({ model: "claude-haiku-5-5", max_tokens: 800,
        system: "Sen bir radyo, TV ve santral spotu metin yazarısın. Türkçe, kısa, akıcı, sesli okunmaya uygun metinler yazarsın. Süre istenirse saniyeye göre kelime sayısını ayarla (30 sn ≈ 70-80 kelime). Yalnızca spot metnini ve gerekirse tek cümlelik not ver.",
        messages: (mesajlar || []).slice(-12) }),
    });
    const d = await r.json(); return j({ yanit: d?.content?.[0]?.text ?? "Yanıt alınamadı." });
  } catch (e) { return j({ hata: "Yapay zeka şu an yanıt veremedi." }); }
});
function j(o: unknown) { return new Response(JSON.stringify(o), { headers: { ...CORS, "content-type": "application/json" } }); }
