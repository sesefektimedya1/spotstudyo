// Gizli anahtar: ANTHROPIC_API_KEY  (Supabase > Edge Functions > Secrets)
// Önce yz-kurulum.txt (SQL) çalıştırılmalı. Adım adım rehber: YZ-KURULUM-REHBERI.txt
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const MODEL = "claude-haiku-5-5";   // küçük ve ucuz model; değiştirmek için yalnızca bu satır
const GUNLUK_LIMIT = 40;            // kişi başı günlük mesaj sınırı (premium ve admin için)
const MAX_KARAKTER = 1500;          // bir mesajın en fazla uzunluğu

const CORS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type" };

const ORTAK = "Sen Türkiye'de radyo, TV ve telefon santrali için çalışan deneyimli bir metin yazarısın. Yalnızca Türkçe yazarsın. Metinler sesli okunmaya uygun, kısa, akıcı ve doğal olmalı; rakamları ve kısaltmaları okunacağı gibi yaz. Cevabında önce hazır metni ver; gerekiyorsa en sonda tek cümlelik not ekle. Kullanıcı bilgi vermediyse uydurma telefon numarası, adres ya da fiyat yazma; yer tutucu olarak köşeli parantez kullan, örn. [firma adı].";
const TUR: Record<string, string> = {
  reklam: "Görev: radyo/TV reklam spotu metni. Süre istenirse saniyeye göre kelime sayısını ayarla (10 sn ≈ 25, 20 sn ≈ 50, 30 sn ≈ 75 kelime). Dikkat çekici bir giriş, tek bir net mesaj ve sonda çağrı (arayın, ziyaret edin) olsun.",
  jingle: "Görev: jingle sözü. Kısa, ritimli, kafiyeli ve akılda kalıcı yaz. Firma adı nakaratta geçsin. Biçim: 'Nakarat:' ve gerekirse 'Kıta:' başlıkları. Müzik yapma, yalnızca söz yaz.",
  santral: "Görev: telefon santrali anonsu (karşılama, mesai dışı, bekleme, yönlendirme). Resmî ama sıcak, kısa ve net olsun. Seçenekleri 'satış için 1'e basın' biçiminde okunur yaz.",
};

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: CORS });
  try {
    const auth = req.headers.get("Authorization");
    if (!auth) return j({ hata: "Giriş gerekli." });
    const sb = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_ANON_KEY")!, { global: { headers: { Authorization: auth } } });
    const { data: u } = await sb.auth.getUser();
    if (!u?.user) return j({ hata: "Giriş gerekli." });

    const { data: p } = await sb.from("profiles").select("role,plan").eq("id", u.user.id).maybeSingle();
    if (!p || (p.role !== "admin" && p.plan !== "premium")) return j({ hata: "Yapay zeka modu ücretli pakete dahildir." });

    // günlük sınır
    const bugun = new Date().toISOString().slice(0, 10);
    const { data: k } = await sb.from("yz_kullanim").select("adet").eq("user_id", u.user.id).eq("gun", bugun).maybeSingle();
    const adet = k?.adet ?? 0;
    if (p.role !== "admin" && adet >= GUNLUK_LIMIT) return j({ hata: "Bugünkü yapay zeka hakkınız doldu. Yarın tekrar deneyin." });

    const govde = await req.json();
    const tur = TUR[govde?.tur] ? govde.tur : "reklam";
    const mesajlar = (Array.isArray(govde?.mesajlar) ? govde.mesajlar : [])
      .filter((m: any) => (m?.role === "user" || m?.role === "assistant") && typeof m?.content === "string")
      .slice(-10)
      .map((m: any) => ({ role: m.role, content: String(m.content).slice(0, MAX_KARAKTER) }));
    if (!mesajlar.length || mesajlar[mesajlar.length - 1].role !== "user") return j({ hata: "Mesaj boş." });

    const r = await fetch("https://api.anthropic.com/v1/messages", {
      method: "POST",
      headers: { "x-api-key": Deno.env.get("ANTHROPIC_API_KEY")!, "anthropic-version": "2023-06-01", "content-type": "application/json" },
      body: JSON.stringify({ model: MODEL, max_tokens: 700, system: ORTAK + "\n\n" + TUR[tur], messages: mesajlar }),
    });
    if (!r.ok) return j({ hata: "Yapay zeka şu an yanıt veremedi." });
    const d = await r.json();
    const yanit = d?.content?.[0]?.text;
    if (!yanit) return j({ hata: "Yanıt alınamadı." });

    await sb.from("yz_kullanim").upsert({ user_id: u.user.id, gun: bugun, adet: adet + 1 });
    return j({ yanit });
  } catch (_e) {
    return j({ hata: "Yapay zeka şu an yanıt veremedi." });
  }
});
function j(o: unknown) { return new Response(JSON.stringify(o), { headers: { ...CORS, "content-type": "application/json" } }); }
